⚠️  CRITICAL REMINDERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. BACKUP YOUR KEYSTORE
	 ► Store android/app/waypoint.keystore in a SECURE location
	 ► Encrypted cloud storage or USB drive recommended
	 ► WITHOUT this keystore, you CANNOT update your app on Play Store!

  2. PROTECT YOUR CREDENTIALS
	 ► Never commit keystore or gradle.properties to git
	 ► Already protected in .gitignore
	 ► Treat passwords like root credentials

  3. USE ONE KEYSTORE FOR ALL UPDATES
	 ► Same keystore = same app signature
	 ► Different keystore = unpublishable update
	 ► Plan for long-term storage

  4. ROTATE PASSWORDS PERIODICALLY
	 ► Change in Windows Credential Manager
	 ► Update environment variables
	 ► Document changes securely

📞 TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Problem: "Keystore not found"
  → Run: ./scripts/create-keystore.ps1

  Problem: "Invalid password"
  → Run: ./scripts/setup-env-vars.ps1

  Problem: "keytool not found"
  → Ensure Java JDK is installed and in PATH
  → Run: keytool -version

  Problem: "Build fails with signing error"
  → Verify env vars: $env:RELEASE_STORE_PASSWORD
  → Run: ./scripts/verify-signing-setup.ps1

  Problem: "gradle.properties not found"
  → File exists in android/ directory
  → Check it's not in .gitignore when needed
